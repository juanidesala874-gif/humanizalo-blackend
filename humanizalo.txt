const express = require('express');
const cors = require('cors');

const app = express();
app.use(cors());
app.use(express.json());

app.post('/analizar', async (req, res) => {
  const { texto } = req.body;
  if (!texto) return res.status(400).json({ error: 'Falta el texto' });

  const chunks = splitChunks(texto);

  const prompt = `Sos un detector experto de texto generado por IA. Analizá el texto y devolvé SOLO JSON válido, sin backticks, sin texto antes ni después.

JSON requerido:
{
  "score": <0-100, nivel de IA del texto completo>,
  "signals": {
    "repeticion": <0-100, alto = mucha repetición = más IA>,
    "conectores": <0-100, alto = conectores genéricos = más IA>,
    "variedad": <0-100, alto = mucha variedad = más humano>,
    "fluidez": <0-100, alto = muy fluido = más humano>
  },
  "verdict": "<Muy humano|Mayormente humano|Mixto|Mayormente IA|Claramente IA>",
  "summary": "<diagnóstico en 8 palabras máximo>",
  "segments": [
    {"text": "<fragmento tal cual>", "score": <0-100>}
  ]
}

Fragmentos a analizar individualmente (${chunks.length} total):
${chunks.map((c, i) => `[${i + 1}] "${c}"`).join('\n')}

Texto completo:
"""
${texto.slice(0, 1400)}
"""`;

  try {
    const data = await callClaude(prompt, 900);
    const clean = data.replace(/```json|```/g, '').trim();
    const json = JSON.parse(clean.slice(clean.indexOf('{'), clean.lastIndexOf('}') + 1));
    if (!json.segments || json.segments.length === 0) {
      json.segments = chunks.map(t => ({ text: t, score: json.score }));
    }
    res.json(json);
  } catch (e) {
    res.status(500).json({ error: e.message });
  }
});

app.post('/humanizar', async (req, res) => {
  const { texto, tono } = req.body;
  if (!texto) return res.status(400).json({ error: 'Falta el texto' });

  const toneMap = {
    natural: 'Reescribí el texto para que suene completamente natural, como si lo escribiera una persona real. Usá variedad léxica y conectores auténticos. Evitá: "fundamental", "crucial", "cabe destacar", "en definitiva", "en conclusión", "es importante mencionar".',
    casual: 'Reescribí en tono relajado y conversacional, como explicándole algo a un amigo. Frases cortas, expresiones cotidianas.',
    profesional: 'Reescribí en tono profesional, claro y directo. Sin frases relleno ni palabras vacías de IA.',
    academico: 'Reescribí con tono académico riguroso pero genuino. Vocabulario técnico apropiado, estructura de oraciones variada.',
    creativo: 'Reescribí con tono creativo y expresivo. Metáforas, ritmo variado, voz propia y distintiva.'
  };

  const instruccion = toneMap[tono] || toneMap.natural;

  const prompt = `${instruccion}

Reglas estrictas:
- No agregues ni quites información del original.
- No uses encabezados ni listas salvo que estén en el original.
- Devolvé SOLO el texto reescrito, sin explicaciones ni comentarios.
- Mantené extensión similar al original.

Texto a reescribir:
${texto}`;

  try {
    const resultado = await callClaude(prompt, 1400);

    const scoreRaw = await callClaude(
      `Qué tan generado por IA suena este texto del 0 al 100. Respondé ÚNICAMENTE con el número entero, nada más.\n\n${resultado.slice(0, 700)}`,
      8
    );
    const nuevoScore = Math.max(0, Math.min(100, parseInt(scoreRaw.trim()) || 10));

    res.json({ resultado, nuevoScore });
  } catch (e) {
    res.status(500).json({ error: e.message });
  }
});

async function callClaude(prompt, maxTokens) {
  const response = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': process.env.ANTHROPIC_API_KEY,
      'anthropic-version': '2023-06-01'
    },
    body: JSON.stringify({
      model: 'claude-sonnet-4-20250514',
      max_tokens: maxTokens,
      messages: [{ role: 'user', content: prompt }]
    })
  });
  if (!response.ok) {
    const err = await response.text();
    throw new Error(`Anthropic error ${response.status}: ${err}`);
  }
  const data = await response.json();
  return data.content.map(b => b.text || '').join('');
}

function splitChunks(text) {
  const sentences = text.match(/[^.!?\n]+[.!?\n]+/g) || [text];
  const chunks = [];
  let cur = '';
  sentences.forEach(s => {
    if ((cur + s).length > 300 && cur) { chunks.push(cur.trim()); cur = s; }
    else cur += s;
  });
  if (cur.trim()) chunks.push(cur.trim());
  return chunks.slice(0, 7);
}

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Humanizalo backend corriendo en puerto ${PORT}`));
